const express = require('express'); const app = express(); app.use(express.json()); app.post('/avisos', (req, res) => { const novoAviso = req.body; res.status(201).json({ mensagem: 'Aviso recebido com sucesso', aviso: novoAviso }); }); app.post('/tarefas', (req, res) => { const novaTarefa = req.body; res.status(201).json({ mensagem: 'Tarefa recebida com sucesso', tarefa: novaTarefa }); }); app.post('/usuarios', (req, res) => { const novoUsuario = req.body; res.status(201).json({ mensagem: 'Usuário recebido com sucesso', usuario: novoUsuario }); }); app.post('/reservas', (req, res) => { const novaReserva = req.body; res.status(201).json({ mensagem: 'Reserva recebida com sucesso', reserva: novaReserva }); }); app.post('/teste-campos', (req, res) => { const dados = req.body; res.status(201).json({ mensagem: 'Objeto com três campos recebido', dados: { campo1: dados.campo1, campo2: dados.campo2, campo3: dados.campo3 } }); }); app.listen(3000, () => { console.log('Servidor rodando'); });
